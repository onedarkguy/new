import React, { useState } from 'react';
import '../allforms.css';

const ContactForm = () => {
 
  const [formData, setFormData] = useState({
    fullName: '',
    email: '',
    topic: 'General Inquiry',
    message: ''
  });

 
  const [submittedData, setSubmittedData] = useState(null);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData({ ...formData, [name]: value });
  };

  const handleSubmit = (e) => {
    e.preventDefault();

    setSubmittedData(formData);
  };

  return (
    <div style={{ marginBottom: '40px', padding: '20px', border: '1px solid #ddd' }}>
      <h2>Contact Form</h2>
      <form onSubmit={handleSubmit}>
        <div>
          <label>Full Name:</label><br />
          <input name="fullName" value={formData.fullName} onChange={handleChange} />
        </div><br />

        <div>
          <label>Email:</label><br />
          <input name="email" type="email" value={formData.email} onChange={handleChange} />
        </div><br />

        <div>
          <label>Topic:</label><br />
          <select name="topic" value={formData.topic} onChange={handleChange}>
            <option value="General Inquiry">General Inquiry</option>
            <option value="Feedback">Feedback</option>
            <option value="Support">Support</option>
          </select>
        </div><br />

        <div>
          <label>Message:</label><br />
          <textarea name="message" value={formData.message} onChange={handleChange} />
        </div><br />

        <button type="submit">Submit</button>
      </form>

      {submittedData && (
        <div style={{ marginTop: '20px', padding: '10px', backgroundColor: '#f4f4f400' }}>
          <h3>Submitted Details:</h3>
          <p><strong>Name:</strong> {submittedData.fullName}</p>
          <p><strong>Email:</strong> {submittedData.email}</p>
          <p><strong>Topic:</strong> {submittedData.topic}</p>
          <p><strong>Message:</strong> {submittedData.message}</p>
        </div>
      )}
    </div>
  );
};

export default ContactForm;
